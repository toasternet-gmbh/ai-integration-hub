bolt Quick Start

Use your API key to authenticate requests with the official Node.js SDK. Ensure you keep your key secure and do not expose it in client-side code.

app.js 
import { HubClient } from '@ai-integration/hub';

const client = new HubClient({
  apiKey: process.env.HUB_API_KEY // Ensure key is in env vars
});

async function fetchRecentOrders() {
  try {
    const orders = await client.tools.orders.get({
      status: 'pending_approval',
      limit: 10
    });
    console.log(orders);
  } catch (error) {
    console.error('Failed to fetch orders:', error);
  }
}

fetchRecentOrders();
